    -m pip install --quiet --disable-pip-version-check --no-deps --no-build-isolation

    OUTPUT "${MIM_PY_VENV_STAMP}"

# ---------------------------------------------------------------------------
# Per-header binding generation → ${CMAKE_CURRENT_BINARY_DIR}/auto_bindings/
# ---------------------------------------------------------------------------
set(_auto_bindings_dir "${CMAKE_CURRENT_BINARY_DIR}/auto_bindings")
set(_binding_script    "${CMAKE_SOURCE_DIR}/scripts/gen_nanobind_bindings.py")
set(_nbextra_dir       "${CMAKE_CURRENT_SOURCE_DIR}/nbextra")

file(MAKE_DIRECTORY "${_auto_bindings_dir}")

# Headers whose bindings are fully autogenerated. Each entry is
# "<header>|<namespace>"; the init function is named init_<header-stem> and the
# generated TU is written to the build tree as auto_bindings/<header-stem>.cpp.
# Order matters for the generated `#include`s and, more importantly, for the
# registration order below: a `Def` subclass must be registered after `Def`, so
# every header defining `Def` subclasses (def, lam, tuple, axm, lattice, check,
# rule) follows def.h.
set(_gen_units
    "${CMAKE_SOURCE_DIR}/submodules/fe/include/fe/sym.h|fe"
    "${CMAKE_SOURCE_DIR}/include/mim/def.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/lam.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/tuple.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/axm.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/lattice.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/check.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/rule.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/world.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/driver.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/flags.h|mim"
    "${CMAKE_SOURCE_DIR}/include/mim/util/log.h|mim"
)

# Every generated unit needs the *whole* manifest: a type is only legal in a
# generated signature if some unit registers it as a Python type, so the
# generator has to know all of them to tell `Def*` (bound) from `mim::Dbg` (not).
# That also makes each unit depend on every manifest header — moving a class
# between two of them changes what the other units may mention.
set(_bound_headers ${_gen_units})
list(TRANSFORM _bound_headers REPLACE "\\|.*$" "")  # drop the |<namespace> half

set(_bound_header_args)
foreach(_header IN LISTS _bound_headers)
    list(APPEND _bound_header_args --bound-header "${_header}")
endforeach()

set(_generated_sources)
foreach(_unit IN LISTS _gen_units)
    string(REPLACE "|" ";" _parts "${_unit}")
    list(GET _parts 0 header)
    list(GET _parts 1 _ns)
    get_filename_component(stem "${header}" NAME_WE)

    set(output "${_auto_bindings_dir}/${stem}.cpp")
    list(APPEND _generated_sources "${output}")

    set(_extra_args)
    set(_extra_deps)
    if(EXISTS "${_nbextra_dir}/${stem}.nbextra")
        set(_extra_args --extra-dir "${_nbextra_dir}")
        list(APPEND _extra_deps "${_nbextra_dir}/${stem}.nbextra")
    endif()

    add_custom_command(
        OUTPUT "${output}"
        COMMAND "${MIM_PY_PYTHON}" "${_binding_script}"
                "${header}"
                --namespace "${_ns}"
                --build-dir "${CMAKE_BINARY_DIR}"
                ${_bound_header_args}
                ${_extra_args}
                -o "${output}"
        DEPENDS
            # Depend on the venv *target*, not on ${MIM_PY_VENV_STAMP}: see mim_py_stubs below.
            mim_py_venv
            ${_bound_headers}
            "${_binding_script}"
            "${CMAKE_BINARY_DIR}/compile_commands.json"
            ${_extra_deps}
        COMMENT "Generating nanobind bindings for ${stem}.h"
        VERBATIM
    )
endforeach()

# Handwritten bindings for surface the generator can't (yet) produce:
#   error - a special nb::exception registration;
#   ast   - ~50 AST classes with Ptr<> (unique_ptr) returns;
#   parser- bespoke plugin() composition logic.
# They must not define an init_* that a generated unit also defines.
set(_handwritten_sources
    "${CMAKE_CURRENT_SOURCE_DIR}/bindings/error.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/bindings/ast.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/bindings/parser.cpp"
)

# The module entry point (NB_MODULE) is autogenerated from an ordered manifest
# of every init function to compose — generated and handwritten alike — so it can
# never drift out of sync with the set of bindings that actually exist.
# Order matters: foundational types are registered first so that later method
# signatures render with Python class names instead of raw C++ spellings.
set(_module_units
    "fe#init_sym"
    "mim#init_def"
    "mim#init_lam"
    "mim#init_tuple"
    "mim#init_axm"
    "mim#init_lattice"
    "mim#init_check"
    "mim#init_rule"
    "mim#init_world"
    "mim#init_flags"
    "mim#init_log"
    "mim#init_driver"
    "mim::ast#init_ast"
    "mim::ast#init_parser"
    "mim#register_error"
)

set(_module_unit_args)
foreach(_u IN LISTS _module_units)
    list(APPEND _module_unit_args --unit "${_u}")
endforeach()

set(_module_source "${_auto_bindings_dir}/py.cpp")
add_custom_command(
    OUTPUT "${_module_source}"
    COMMAND "${MIM_PY_PYTHON}" "${_binding_script}"
            --emit-module _mim
            ${_module_unit_args}
            -o "${_module_source}"
    DEPENDS
        mim_py_venv
        "${_binding_script}"
    COMMENT "Generating nanobind module entry point py.cpp"
    VERBATIM
)

nanobind_add_module(pymim
    "${_module_source}"
    ${_handwritten_sources}
    ${_generated_sources}
)

set_target_properties(pymim PROPERTIES LIBRARY_OUTPUT_NAME _mim)
target_link_libraries(pymim PRIVATE libmim)
# for nb_type_hook.h, which every generated unit includes
target_include_directories(pymim PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

# Generate a stub for the internal nanobind module.
set(MIM_PY_STUBS_DIR "${CMAKE_BINARY_DIR}/mim_py_stubs")
set(MIM_PY_STUB_FILE "${MIM_PY_STUBS_DIR}/_mim.pyi")
set(MIM_PY_NANOBIND_STUBGEN "${CMAKE_SOURCE_DIR}/submodules/nanobind/src/stubgen.py")
set(_mim_py_stub_runtime_commands)
if(WIN32)
    set(_mim_py_stub_runtime_commands
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                $<TARGET_RUNTIME_DLLS:pymim>
                "${MIM_PY_STUBS_DIR}"
    )
endif()
add_custom_command(
    OUTPUT "${MIM_PY_STUB_FILE}"
    COMMAND "${CMAKE_COMMAND}" -E rm -rf "${MIM_PY_STUBS_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${MIM_PY_STUBS_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E copy
            "$<TARGET_FILE:pymim>"
            "${MIM_PY_STUBS_DIR}/$<TARGET_FILE_NAME:pymim>"
    ${_mim_py_stub_runtime_commands}
    COMMAND "${CMAKE_COMMAND}" -E env
            "PYTHONPATH=${MIM_PY_STUBS_DIR}"
            "${MIM_PY_PYTHON}" "${MIM_PY_NANOBIND_STUBGEN}"
                -q
                -m _mim
                -o "${MIM_PY_STUB_FILE}"
    # Depend on the venv *target*, not on ${MIM_PY_VENV_STAMP}: the Makefile generator
    # duplicates a custom command into every target that consumes its output file, so a
    # file-level dependency would let "make -j" run two venv creations at the same time.
    DEPENDS pymim mim_py_venv "${MIM_PY_NANOBIND_STUBGEN}"
    COMMENT "mim_py: generating _mim.pyi via nanobind stubgen"
    COMMAND_EXPAND_LISTS
    VERBATIM
)
add_custom_target(mim_py_stubs DEPENDS "${MIM_PY_STUB_FILE}")

# Stage the package (sources + generated artifacts) and install it editable.
# Editable install lets later builds refresh staged files without re-running pip.
set(_mim_py_plugin_stage "${MIM_PY_STAGE}/src/mim/_plugins")
set(_mim_py_main_runtime_commands)
if(WIN32)
    set(_mim_py_main_runtime_commands
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                $<TARGET_RUNTIME_DLLS:pymim>
                "${MIM_PY_STAGE}/src/mim"
    )
endif()
set(_mim_py_stage_plugin_commands
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${_mim_py_plugin_stage}"
    COMMAND "${CMAKE_COMMAND}" -E touch "${_mim_py_plugin_stage}/__init__.py"
)
foreach(_plugin ${MIM_PLUGIN_LIST})
    list(APPEND _mim_py_stage_plugin_commands
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                "${CMAKE_BINARY_DIR}/${CMAKE_INSTALL_LIBDIR}/mim/${_plugin}.py"
                "${_mim_py_plugin_stage}/${_plugin}.py"
    )
endforeach()
add_custom_target(mim_py_main
    # Wipe the source side of the stage so deleted py/src files don't linger.
    # Generated artifacts under src/mim/_plugins are restaged below.
    COMMAND "${CMAKE_COMMAND}" -E rm -rf "${MIM_PY_STAGE}/src" "${MIM_PY_STAGE}/tests"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}" "${MIM_PY_STAGE}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<TARGET_FILE:pymim>"
            "${MIM_PY_STAGE}/src/mim/$<TARGET_FILE_NAME:pymim>"
    ${_mim_py_main_runtime_commands}
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${MIM_PY_STUB_FILE}"
            "${MIM_PY_STAGE}/src/mim/_mim.pyi"
    ${_mim_py_stage_plugin_commands}
    COMMAND ${_mim_py_python_command} ${_mim_py_pip_install} -e "${MIM_PY_STAGE}"
    COMMENT "mim_py: staging & installing mim (editable)"
    COMMAND_EXPAND_LISTS
    VERBATIM
)
add_dependencies(mim_py_main pymim mim_all_plugins mim_py_venv mim_py_stubs)

# Editable installs for third-party plugins under py/plugins/*/pyproject.toml.
file(GLOB _mim_py_plugin_tomls CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_SOURCE_DIR}/plugins/*/pyproject.toml")
set(_mim_py_plugin_targets)
foreach(_toml ${_mim_py_plugin_tomls})
    get_filename_component(_plugin_dir  "${_toml}" DIRECTORY)
    get_filename_component(_plugin_name "${_plugin_dir}" NAME)
    set(_target mim_py_plugin_${_plugin_name})
    set(_stage "${CMAKE_BINARY_DIR}/mim_py_stage/plugins/${_plugin_name}")
    add_custom_target(${_target}
        COMMAND "${CMAKE_COMMAND}" -E copy_directory "${_plugin_dir}" "${_stage}"
        COMMAND ${_mim_py_python_command} ${_mim_py_pip_install} -e "${_stage}"
        COMMENT "mim_py: staging & installing ${_target} (editable)"
        VERBATIM
    )
    add_dependencies(${_target} mim_py_main)
    list(APPEND _mim_py_plugin_targets ${_target})
endforeach()

add_custom_target(mim_py ALL)
add_dependencies(mim_py mim_py_main ${_mim_py_plugin_targets})
