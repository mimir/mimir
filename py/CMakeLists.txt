# Importable in place via PYTHONPATH; `cmake --install` copies this tree into site-packages.
set(MIM_PY_STAGE "${CMAKE_BINARY_DIR}/python")
set(_mim_py_pkg "${MIM_PY_STAGE}/mim")

set(_auto_bindings_dir "${CMAKE_CURRENT_BINARY_DIR}/auto_bindings")
set(_binding_script    "${CMAKE_SOURCE_DIR}/scripts/gen_nanobind_bindings.py")
set(_nbextra_dir       "${CMAKE_CURRENT_SOURCE_DIR}/nbextra")
set(_log_dir           "${CMAKE_CURRENT_BINARY_DIR}/log")

file(MAKE_DIRECTORY "${_auto_bindings_dir}")

# The generator reports every dropped member on every run, which drowns out the
# rest of the build; each unit's report goes to ${_log_dir}/<stem>.log instead.
set(_run_logged "${CMAKE_COMMAND}" -P "${CMAKE_SOURCE_DIR}/cmake/run-logged.cmake" --)

# Headers whose bindings are fully autogenerated. The init function is named
# init_<header-stem>, the generated TU is written to the build tree as
# auto_bindings/<header-stem>.cpp, and the C++ namespace mirrors the include
# directory the header sits in.
# Order matters for the generated `#include`s and, more importantly, for the
# registration order below: a `Def` subclass must be registered after `Def`, so
# every header defining `Def` subclasses (def, lam, tuple, axm, single, union, nom,
# check, rule) follows def.h.
set(_gen_headers
    "${CMAKE_SOURCE_DIR}/submodules/fe/include/fe/sym.h"
    "${CMAKE_SOURCE_DIR}/include/mim/def.h"
    "${CMAKE_SOURCE_DIR}/include/mim/lam.h"
    "${CMAKE_SOURCE_DIR}/include/mim/tuple.h"
    "${CMAKE_SOURCE_DIR}/include/mim/axm.h"
    "${CMAKE_SOURCE_DIR}/include/mim/single.h"
    "${CMAKE_SOURCE_DIR}/include/mim/union.h"
    "${CMAKE_SOURCE_DIR}/include/mim/nom.h"
    "${CMAKE_SOURCE_DIR}/include/mim/check.h"
    "${CMAKE_SOURCE_DIR}/include/mim/rule.h"
    "${CMAKE_SOURCE_DIR}/include/mim/world.h"
    "${CMAKE_SOURCE_DIR}/include/mim/driver.h"
    "${CMAKE_SOURCE_DIR}/include/mim/flags.h"
    "${CMAKE_SOURCE_DIR}/submodules/fe/include/fe/log.h"
)

# Every generated unit needs the *whole* manifest: a type is only legal in a
# generated signature if some unit registers it as a Python type, so the
# generator has to know all of them to tell `Def*` (bound) from `mim::Dbg` (not).
# That also makes each unit depend on every manifest header — moving a class
# between two of them changes what the other units may mention.
set(_bound_header_args)
foreach(header IN LISTS _gen_headers)
    list(APPEND _bound_header_args --bound-header "${header}")
endforeach()

set(_generated_sources)
foreach(header IN LISTS _gen_headers)
    cmake_path(GET header PARENT_PATH dir)
    cmake_path(GET dir FILENAME ns)
    cmake_path(GET header STEM stem)

    set(output "${_auto_bindings_dir}/${stem}.cpp")
    list(APPEND _generated_sources "${output}")

    set(_extra_args)
    set(_extra_deps)
    if(EXISTS "${_nbextra_dir}/${stem}.nbextra")
        set(_extra_args --extra-dir "${_nbextra_dir}")
        list(APPEND _extra_deps "${_nbextra_dir}/${stem}.nbextra")
    endif()

    add_custom_command(
        OUTPUT "${output}"
        COMMAND ${_run_logged} "${_log_dir}/${stem}.log"
                "${Python_EXECUTABLE}" "${_binding_script}"
                "${header}"
                --namespace "${ns}"
                --build-dir "${CMAKE_BINARY_DIR}"
                ${_bound_header_args}
                ${_extra_args}
                -o "${output}"
        DEPENDS
            ${_gen_headers}
            "${_binding_script}"
            "${CMAKE_BINARY_DIR}/compile_commands.json"
            ${_extra_deps}
        COMMENT "Generating nanobind bindings for ${stem}.h"
        VERBATIM
    )
endforeach()

# Handwritten bindings for surface the generator can't (yet) produce:
#   error - a special nb::exception registration;
#   ast   - ~50 AST classes with Ptr<> (unique_ptr) returns;
#   parser- bespoke plugin() composition logic.
# They must not define an init_* that a generated unit also defines.
set(_handwritten_sources
    "${CMAKE_CURRENT_SOURCE_DIR}/bindings/error.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/bindings/ast.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/bindings/parser.cpp"
)

# The module entry point (NB_MODULE) is autogenerated from an ordered manifest
# of every init function to compose — generated and handwritten alike — so it can
# never drift out of sync with the set of bindings that actually exist.
# Order matters: foundational types are registered first so that later method
# signatures render with Python class names instead of raw C++ spellings.
set(_module_units
    "fe#init_sym"
    "mim#init_def"
    "mim#init_lam"
    "mim#init_tuple"
    "mim#init_axm"
    "mim#init_single"
    "mim#init_union"
    "mim#init_nom"
    "mim#init_check"
    "mim#init_rule"
    "mim#init_world"
    "mim#init_flags"
    "fe#init_log"
    "mim#init_driver"
    "mim::ast#init_ast"
    "mim::ast#init_parser"
    "mim#register_error"
)

set(_module_unit_args)
foreach(_u IN LISTS _module_units)
    list(APPEND _module_unit_args --unit "${_u}")
endforeach()

set(_module_source "${_auto_bindings_dir}/py.cpp")
add_custom_command(
    OUTPUT "${_module_source}"
    COMMAND ${_run_logged} "${_log_dir}/py.log"
            "${Python_EXECUTABLE}" "${_binding_script}"
            --emit-module _mim
            ${_module_unit_args}
            -o "${_module_source}"
    DEPENDS
        "${_binding_script}"
    COMMENT "Generating nanobind module entry point py.cpp"
    VERBATIM
)

nanobind_add_module(pymim
    "${_module_source}"
    ${_handwritten_sources}
    ${_generated_sources}
)

set_target_properties(pymim PROPERTIES LIBRARY_OUTPUT_NAME _mim)
target_link_libraries(pymim PRIVATE libmim)
# for nb_type_hook.h, which every generated unit includes
target_include_directories(pymim PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

# Generate a stub for the internal nanobind module.
set(MIM_PY_STUBS_DIR "${CMAKE_BINARY_DIR}/mim_py_stubs")
set(MIM_PY_STUB_FILE "${MIM_PY_STUBS_DIR}/_mim.pyi")
set(MIM_PY_NANOBIND_STUBGEN "${CMAKE_SOURCE_DIR}/submodules/nanobind/src/stubgen.py")
set(_mim_py_stub_runtime_commands)
if(WIN32)
    set(_mim_py_stub_runtime_commands
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                $<TARGET_RUNTIME_DLLS:pymim>
                "${MIM_PY_STUBS_DIR}"
    )
endif()
add_custom_command(
    OUTPUT "${MIM_PY_STUB_FILE}"
    COMMAND "${CMAKE_COMMAND}" -E rm -rf "${MIM_PY_STUBS_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${MIM_PY_STUBS_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E copy
            "$<TARGET_FILE:pymim>"
            "${MIM_PY_STUBS_DIR}/$<TARGET_FILE_NAME:pymim>"
    ${_mim_py_stub_runtime_commands}
    COMMAND "${CMAKE_COMMAND}" -E env
            "PYTHONPATH=${MIM_PY_STUBS_DIR}"
            "${Python_EXECUTABLE}" "${MIM_PY_NANOBIND_STUBGEN}"
                -q
                -m _mim
                -o "${MIM_PY_STUB_FILE}"
    DEPENDS pymim "${MIM_PY_NANOBIND_STUBGEN}"
    COMMENT "mim_py: generating _mim.pyi via nanobind stubgen"
    COMMAND_EXPAND_LISTS
    VERBATIM
)
add_custom_target(mim_py_stubs DEPENDS "${MIM_PY_STUB_FILE}")

file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/py/mim/_plugins")
file(TOUCH "${CMAKE_BINARY_DIR}/py/mim/_plugins/__init__.py")

set(_mim_py_runtime_commands)
if(WIN32)
    set(_mim_py_runtime_commands
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                $<TARGET_RUNTIME_DLLS:pymim>
                "${_mim_py_pkg}"
    )
endif()
add_custom_target(mim_py ALL
    # Wipe the stage so deleted py/src files don't linger.
    COMMAND "${CMAKE_COMMAND}" -E rm -rf "${_mim_py_pkg}"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/src/mim" "${_mim_py_pkg}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:pymim>" "${_mim_py_pkg}"
    ${_mim_py_runtime_commands}
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${MIM_PY_STUB_FILE}" "${_mim_py_pkg}"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_BINARY_DIR}/py/mim/_plugins" "${_mim_py_pkg}/_plugins"
    COMMENT "mim_py: staging mim in ${MIM_PY_STAGE}"
    COMMAND_EXPAND_LISTS
    VERBATIM
)
add_dependencies(mim_py pymim mim_all_plugins mim_py_stubs)

execute_process(
    COMMAND "${Python_EXECUTABLE}" -c
            "import sysconfig; print(sysconfig.get_path('platlib', vars={'base': '', 'platbase': ''}).lstrip('/\\\\'))"
    OUTPUT_VARIABLE _mim_py_platlib
    OUTPUT_STRIP_TRAILING_WHITESPACE
)
set(MIM_PY_INSTALL_DIR "${_mim_py_platlib}" CACHE STRING "Install directory of the mim Python package, relative to the install prefix.")
set(_mim_py_install_pkg "${MIM_PY_INSTALL_DIR}/mim")

cmake_path(
    RELATIVE_PATH CMAKE_INSTALL_LIBDIR
    BASE_DIRECTORY "${_mim_py_install_pkg}"
    OUTPUT_VARIABLE _mim_py_rel_libdir
)
if(APPLE)
    set_target_properties(pymim PROPERTIES INSTALL_RPATH "@loader_path/${_mim_py_rel_libdir}")
else()
    set_target_properties(pymim PROPERTIES INSTALL_RPATH "$ORIGIN/${_mim_py_rel_libdir}")
endif()

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/pyproject.toml" _mim_py_version REGEX "^version = ")
string(REGEX REPLACE "^version = \"(.*)\"$" "\\1" _mim_py_version "${_mim_py_version}")
file(CONFIGURE
    OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/METADATA"
    CONTENT "Metadata-Version: 2.1\nName: mim\nVersion: ${_mim_py_version}\n"
)

install(
    DIRECTORY "${_mim_py_pkg}/"
    DESTINATION "${_mim_py_install_pkg}"
    # The staged extension and DLLs carry build-tree paths; the install rules below provide relocated ones.
    PATTERN "*.so" EXCLUDE
    PATTERN "*.pyd" EXCLUDE
    PATTERN "*.dll" EXCLUDE
    PATTERN "__pycache__" EXCLUDE
)
install(TARGETS pymim LIBRARY DESTINATION "${_mim_py_install_pkg}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/METADATA" DESTINATION "${MIM_PY_INSTALL_DIR}/mim-${_mim_py_version}.dist-info")

# Windows has no RPATH, so pytest needs the mim DLLs on PATH.
# The ;-separated value has to stay inline: routed through a variable, CMake list
# semantics would split it into separate arguments and `env` would run the 2nd as the command.
if(WIN32)
    add_custom_target(test-py
        COMMAND "${CMAKE_COMMAND}" -E env
                "PATH=${CMAKE_RUNTIME_OUTPUT_DIRECTORY};${CMAKE_BINARY_DIR}/${CMAKE_INSTALL_LIBDIR}/mim;$ENV{PATH}"
                "PYTHONPATH=${MIM_PY_STAGE}"
                "${Python_EXECUTABLE}" -m pytest "${CMAKE_CURRENT_SOURCE_DIR}/tests"
        DEPENDS mim_py
        COMMENT "mim_py: running Python tests"
        USES_TERMINAL
        VERBATIM
    )
else()
    add_custom_target(test-py
        COMMAND "${CMAKE_COMMAND}" -E env "PYTHONPATH=${MIM_PY_STAGE}"
                "${Python_EXECUTABLE}" -m pytest "${CMAKE_CURRENT_SOURCE_DIR}/tests"
        DEPENDS mim_py
        COMMENT "mim_py: running Python tests"
        USES_TERMINAL
        VERBATIM
    )
endif()

if(TARGET test-all)
    add_dependencies(test-all test-py)
endif()
